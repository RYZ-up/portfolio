\mainproject
{Mai 2026}
{Stage assistant ingénieur -- Système de diagnostic électronique assisté par IA}
{Phone Addict, Créteil}
{%
\begin{itemize}
    \item \textbf{Volet hardware, instrumentation :} conception d'une carte de mesure ESP32 (WiFi) pour outiller le diagnostic électronique en atelier : ADC 16~bits ADS1115 en I\textsuperscript{2}C (3 tensions, $\pm$6,144~V, résolution 187,5~\textmu V/LSB), capteur de courant à effet Hall ACS712~5~A (185~mV/A), affichage OLED SSD1306. Statistiques de mesure en temps réel (moyenne, écart-type, tendance) et étude Scilab de l'erreur de quantification de l'ADC pour valider la précision du système.
    \item \textbf{Volet logiciel, IA :} intégration d'un LLM exécuté en local (100\,\% hors ligne, sans dépendance cloud) pour interpréter les mesures et proposer des pistes de panne.
    \item \textbf{Volet logiciel, sortie :} génération automatique de rapports de mesure (PDF) et classification des anomalies par apprentissage supervisé. Rapport de stage : 16,5/20.
\end{itemize}
}

\mainproject

{Chariot robotisé -- Châssis mobile articulé}
{UPEC}
{%
\begin{itemize}
    \item \textbf{Motorisation :} 2 moteurs à courant continu EMG30 à encodeurs, alimentés par batterie et commandés en puissance par un pont en H L298 (PWM, sens de rotation) depuis une Arduino Uno.
    \item \textbf{Logiciel \& mécatronique :} firmware C++ de contrôle moteur exploitant les encodeurs (mesure de rotation) ; bras à 3 degrés de liberté intégré sur châssis articulé ; chaîne de puissance, câblage, intégration mécanique.
\end{itemize}
}

\mainproject
{Déc. 2025}
{Bras robotisé -- Commande par gants de données}
{UPEC}
{%
\begin{itemize}
    \item \textbf{Capteurs :} deux gants instrumentés de centrales inertielles MPU6050 (gyroscope + accéléromètre 3 axes, bus I\textsuperscript{2}C) mesurant l'orientation de la main, acquises par un ESP32.
    \item \textbf{Commande :} modèle de cinématique directe convertissant les angles mesurés en consignes articulaires (signaux PWM) pour 4 servomoteurs et une pince pilotée en complément.
    \item \textbf{Logiciel \& mécanique :} firmware C++ d'acquisition et de commande ; liaison filaire à latence de quelques ms (téléopération temps réel) ; interface PC de supervision et de contrôle ; structure conçue et imprimée en 3D.
\end{itemize}
}
